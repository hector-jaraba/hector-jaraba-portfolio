### JavaScript/TypeScript

Here's a TypeScript example with modern React hooks:

```typescript
import { useState, useEffect } from 'react';

interface User {
  id: string;
  name: string;
  email: string;
  role: 'admin' | 'user' | 'guest';
}

function UserProfile({ userId }: { userId: string }) {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  useEffect(() => {
    async function fetchUser() {
      try {
        setLoading(true);
        const response = await fetch(`/api/users/${userId}`);

        if (!response.ok) {
          throw new Error('Failed to fetch user');
        }

        const data = await response.json();
        setUser(data);
      } catch (err) {
        setError(err instanceof Error ? err : new Error('Unknown error'));
      } finally {
        setLoading(false);
      }
    }

    fetchUser();
  }, [userId]);

  if (loading) return <div>Loading...</div>;
  if (error) return <div>Error: {error.message}</div>;
  if (!user) return <div>User not found</div>;

  return (
    <div className="user-profile">
      <h2>{user.name}</h2>
      <p>{user.email}</p>
      <span className="badge">{user.role}</span>
    </div>
  );
}

export default UserProfile;
```

### Python

Python example with modern features:

```python
from dataclasses import dataclass
from typing import List, Optional
import asyncio

@dataclass
class User:
    id: int
    username: str
    email: str
    is_active: bool = True

    def __post_init__(self):
        if not self.email or '@' not in self.email:
            raise ValueError("Invalid email address")

    def to_dict(self) -> dict:
        return {
            'id': self.id,
            'username': self.username,
            'email': self.email,
            'is_active': self.is_active
        }

class UserService:
    def __init__(self, database):
        self.db = database

    async def get_user(self, user_id: int) -> Optional[User]:
        """Fetch user from database by ID"""
        try:
            data = await self.db.fetch_one(
                "SELECT * FROM users WHERE id = ?",
                (user_id,)
            )

            if data:
                return User(**data)
            return None
        except Exception as e:
            print(f"Error fetching user: {e}")
            raise

    async def get_active_users(self) -> List[User]:
        """Get all active users"""
        data = await self.db.fetch_all(
            "SELECT * FROM users WHERE is_active = TRUE"
        )
        return [User(**row) for row in data]

# Usage
async def main():
    service = UserService(database)
    user = await service.get_user(123)
    if user:
        print(f"Found user: {user.username}")
```

### Custom Hook Pattern

```typescript
import { useState, useEffect, useCallback, useRef } from 'react';

interface UseAsyncOptions<T> {
  immediate?: boolean;
  onSuccess?: (data: T) => void;
  onError?: (error: Error) => void;
}

interface UseAsyncReturn<T, Args extends any[]> {
  execute: (...args: Args) => Promise<void>;
  data: T | null;
  loading: boolean;
  error: Error | null;
  reset: () => void;
}

function useAsync<T, Args extends any[]>(
  asyncFunction: (...args: Args) => Promise<T>,
  options: UseAsyncOptions<T> = {}
): UseAsyncReturn<T, Args> {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);

  const { immediate = false, onSuccess, onError } = options;
  const mountedRef = useRef(true);

  const execute = useCallback(
    async (...args: Args) => {
      setLoading(true);
      setError(null);

      try {
        const result = await asyncFunction(...args);

        if (mountedRef.current) {
          setData(result);
          onSuccess?.(result);
        }
      } catch (err) {
        if (mountedRef.current) {
          const error = err instanceof Error ? err : new Error('Unknown error');
          setError(error);
          onError?.(error);
        }
      } finally {
        if (mountedRef.current) {
          setLoading(false);
        }
      }
    },
    [asyncFunction, onSuccess, onError]
  );

  const reset = useCallback(() => {
    setData(null);
    setError(null);
    setLoading(false);
  }, []);

  useEffect(() => {
    mountedRef.current = true;
    return () => {
      mountedRef.current = false;
    };
  }, []);

  useEffect(() => {
    if (immediate) {
      execute();
    }
  }, [immediate, execute]);

  return { execute, data, loading, error, reset };
}

// Usage example
function UserList() {
  const { data, loading, error, execute } = useAsync(
    async (query: string) => {
      const response = await fetch(`/api/users?q=${query}`);
      return response.json();
    },
    {
      onSuccess: (users) => console.log('Loaded users:', users),
      onError: (error) => console.error('Failed to load:', error),
    }
  );

  return (
    <div>
      <button onClick={() => execute('admin')}>
        Load Admin Users
      </button>
      {loading && <p>Loading...</p>}
      {error && <p>Error: {error.message}</p>}
      {data && <UserGrid users={data} />}
    </div>
  );
}
```

### Memoization Example

```typescript
import { memo, useMemo, useCallback } from 'react';

interface Item {
  id: string;
  name: string;
  price: number;
  quantity: number;
}

interface ShoppingCartProps {
  items: Item[];
  onUpdateQuantity: (id: string, quantity: number) => void;
  onRemoveItem: (id: string) => void;
}

const ShoppingCart = memo<ShoppingCartProps>(({
  items,
  onUpdateQuantity,
  onRemoveItem
}) => {
  // Memoize expensive calculations
  const totals = useMemo(() => {
    return items.reduce(
      (acc, item) => ({
        items: acc.items + item.quantity,
        price: acc.price + item.price * item.quantity,
      }),
      { items: 0, price: 0 }
    );
  }, [items]);

  // Memoize event handlers to prevent unnecessary re-renders
  const handleQuantityChange = useCallback(
    (id: string, delta: number) => {
      const item = items.find((i) => i.id === id);
      if (item) {
        onUpdateQuantity(id, Math.max(0, item.quantity + delta));
      }
    },
    [items, onUpdateQuantity]
  );

  return (
    <div className="shopping-cart">
      <h2>Shopping Cart</h2>

      <div className="cart-items">
        {items.map((item) => (
          <CartItem
            key={item.id}
            item={item}
            onIncrease={() => handleQuantityChange(item.id, 1)}
            onDecrease={() => handleQuantityChange(item.id, -1)}
            onRemove={() => onRemoveItem(item.id)}
          />
        ))}
      </div>

      <div className="cart-summary">
        <p>Total Items: {totals.items}</p>
        <p>Total Price: ${totals.price.toFixed(2)}</p>
      </div>
    </div>
  );
});

ShoppingCart.displayName = 'ShoppingCart';
```
